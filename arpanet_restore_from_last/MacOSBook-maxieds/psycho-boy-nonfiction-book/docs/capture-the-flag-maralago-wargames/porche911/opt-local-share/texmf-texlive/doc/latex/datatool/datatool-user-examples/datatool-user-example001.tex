\documentclass[12pt]{article}
\pagestyle{empty}
\ExplSyntaxOn
\NewDocumentCommand{\testreplace} { m }
{
 \regex_replace_all:nnN
  { \c{emph} \cB\{ (\w+) \cE\}}
  { \c{textbf} { \c{text_uppercase:n}{ \1 } } }
 #1
}
\ExplSyntaxOff 
\begin{document}
\newcommand{\teststring}{The duck said \emph{boo} to the goose.}
Original: \teststring 

\testreplace{\teststring}
Replaced: \teststring 
\end{document}
