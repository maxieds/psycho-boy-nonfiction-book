\begin{definition}
%Definition~\ref{D:well}
A diagram of an SPS lattice $L$ is a
\emph{${C}_1$-diagram} if the middle edge of any cover-preserving $7$ is steep
and all other edges are normal.
\end{definition}

\begin{theorem}\label{T:well}
%Theorem~\ref{T:well}
Every slim, planar, semimodular lattice $L$ has a ${C}_1$-diagram.
\end{theorem}
