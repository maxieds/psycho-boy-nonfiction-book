\documentclass[12pt]{article}
\pagestyle{empty}
% sample CSV file:
\begin{filecontents}[noheader,overwrite]{fruit.csv}
Name,Quantity
"Apples",30
"Pears",25
"Lemons,Limes",40.5
"Peaches",34.5
"Cherries",20
\end{filecontents}


\usepackage{datapie}
% Load data from fruit.csv file:
\DTLsetup{store-datum,default-name=fruit}
\DTLread{fruit.csv} 
\begin{document}
\DTLpiechart
{
 variable=\Quantity,
 outer-label=\Name,
 cutaway={1-2}
}
{fruit}% database
{\Quantity=Quantity,\Name=Name}% assignment list
 
\end{document}
