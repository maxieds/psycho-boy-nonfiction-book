\newcommand{\DTLisLocaleHandler}[1]{
 \regex_replace_case_all:nN
 {

  { Ð } { D\cL\x{7f} }
  { ð } { d\cL\x{7f} }
  { É } { E\cL\x{7f} }
  { é } { e\cL\x{7f} }
  { Í } { I\cL\x{7f} }
  { í } { i\cL\x{7f} }
  { Ó } { O\cL\x{7f} }
  { ó } { o\cL\x{7f} }
  { Ú } { U\cL\x{7f} }
  { ú } { u\cL\x{7f} }
  { Ý } { Y\cL\x{7f} }
  { ý } { y\cL\x{7f} }
  { Þ } { \cL\x{5b} }

  { æ } { \cL\x{7c} }
  { Ö } { \cL\x{5d} }
  { ö } { \cL\x{7d} }
  { ([[:punct:]]+) } { \cO"\1 }
  }
 #1
}
